// Carga dinámica de títulos async function cargarTitulosDinamicos() { try { const resp = await fetch('titulos.json?v=' + Date.now()); // Evita caché antigua if (resp.ok) { const data = await resp.json(); for (const k in data) { const n = parseInt(k, 10); if (n >= 1 && n <= 613) { titulosOficiales[n] = data[k]; titulosOficiales[String(n)] = data[k]; } } } } catch (e) { console.warn("No se pudo cargar titulos.json local, usando catálogo base.", e); } himnos.length = 0; for (let i = 1; i <= 613; i++) { const numStr = String(i).padStart(3, '0'); const titleStr = titulosOficiales[i] || titulosOficiales[String(i)] || `Himno ${numStr}`; himnos.push({ num: numStr, title: titleStr }); } renderList(getFilteredList()); } // Generador optimizado de URLs rápidas sin reintentos lentos function getHymnCandidateUrls(h, isInstrumental) { if (!h) return []; const numPad = h.num; // "132" const numInt = parseInt(h.num, 10).toString(); // "132" const titleRaw = h.title.trim(); const titleNoAcc = removeAccents(titleRaw); const titleLower = titleNoAcc.toLowerCase(); const titleCleanPunct = cleanPunctuation(titleRaw); const folder = isInstrumental ? 'Instrumental' : 'Cantado'; const baseUrl = `https://files.recursos-biblicos.com/Himnario/mp4/${folder}`; // Variantes de títulos registradas en el servidor const titleVariants = Array.from(new Set([ titleRaw, titleNoAcc, titleLower, titleCleanPunct ])).filter(Boolean); const numVariants = Array.from(new Set([numPad, numInt])); const fileNames = []; numVariants.forEach(nv => { titleVariants.forEach(tv => { if (isInstrumental) { fileNames.push(`${nv} ${tv}_pista.mp4`); fileNames.push(`${nv} ${tv}_PISTA.mp4`); fileNames.push(`${nv} ${tv}.mp4`); } else { fileNames.push(`${nv} ${tv}.mp4`); } }); // Fallbacks por número directo if (isInstrumental) { fileNames.push(`${nv}_pista.mp4`); fileNames.push(`${nv}_PISTA.mp4`); fileNames.push(`${nv}.mp4`); } else { fileNames.push(`${nv}.mp4`); } }); // Construir URLs limpias apuntando exclusivamente al servidor activo const urlsUnicas = Array.from(new Set(fileNames)).map(fn => `${baseUrl}/${encodeURIComponent(fn)}`); return urlsUnicas; }
